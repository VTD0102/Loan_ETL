-- Create core.risk_assessment table
-- Run this in Supabase SQL Editor BEFORE running predict_engine.py
--
-- Uses listing_key (VARCHAR) as unique identifier
-- matching core.loans PRIMARY KEY = listing_key

CREATE TABLE IF NOT EXISTS core.risk_assessment (
    id                      SERIAL PRIMARY KEY,
    listing_key             VARCHAR(100) NOT NULL,
    loan_number             VARCHAR(50),
    probability_of_default  NUMERIC(6, 4),
    risk_score_internal     INTEGER,
    risk_level              VARCHAR(10) CHECK (risk_level IN ('Low', 'Medium', 'High')),
    recommended_amount      NUMERIC(12, 2),
    recommended_term        INTEGER,
    assessment_date         TIMESTAMPTZ,

    -- Unique constraint — one assessment per loan
    CONSTRAINT uq_risk_assessment_listing_key UNIQUE (listing_key),

    -- FK back to core.loans
    CONSTRAINT fk_risk_assessment_loan
        FOREIGN KEY (listing_key)
        REFERENCES core.loans(listing_key)
);

COMMENT ON TABLE core.risk_assessment IS
'ML risk assessment results per loan. Generated by predict_engine.py.';

COMMENT ON COLUMN core.risk_assessment.listing_key IS
'FK to core.loans.listing_key — unique identifier per loan.';

COMMENT ON COLUMN core.risk_assessment.loan_number IS
'Human-readable loan number (VARCHAR) matching core.loans.loan_number.';

COMMENT ON COLUMN core.risk_assessment.probability_of_default IS
'Predicted probability of default from RandomForest model (0.0 - 1.0).';

COMMENT ON COLUMN core.risk_assessment.risk_score_internal IS
'Internal risk score = (1 - probability_of_default) * 100. Higher = safer.';

COMMENT ON COLUMN core.risk_assessment.risk_level IS
'Risk label: Low (<0.2), Medium (0.2-0.4), High (>0.4).';

COMMENT ON COLUMN core.risk_assessment.recommended_amount IS
'Suggested loan amount based on risk level.';

COMMENT ON COLUMN core.risk_assessment.recommended_term IS
'Suggested loan term in months based on risk level.';

-- Index for fast lookup by loan_number (used in prediction_ui.py)
CREATE INDEX IF NOT EXISTS idx_risk_assessment_loan_number
    ON core.risk_assessment(loan_number);

-- Index for filtering by risk_level (used in dashboard)
CREATE INDEX IF NOT EXISTS idx_risk_assessment_risk_level
    ON core.risk_assessment(risk_level);
